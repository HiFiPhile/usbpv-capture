cmake_minimum_required(VERSION 3.16)
project(usbpv_diagnostic_tools LANGUAGES CXX)
set(diagnostic_tools analyze_load)
if(WIN32)
  list(APPEND diagnostic_tools usb_storage_load)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  list(APPEND diagnostic_tools usb_storage_load_linux)
endif()
foreach(name IN LISTS diagnostic_tools)
  add_executable(${name} ${name}.cpp)
  target_compile_features(${name} PRIVATE cxx_std_17)
  if(MINGW)
    target_link_options(${name} PRIVATE -static)
  endif()
endforeach()
